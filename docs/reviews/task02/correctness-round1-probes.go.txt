package memy_test

func TestReviewSweepPolicyBump(t *testing.T){
 f:=newFixture(t,nil); q:=memy.SweepRequest{OperationID:"review-pass",Limit:1,MaxBytes:1<<20}
 a,e:=f.engine.Sweep(t.Context(),f.actor,f.scope,"assist",q);if e!=nil||a.Complete{t.Fatal(a,e)}
 f.policy.Grant("alice",f.scope,"authority/v2",memy.ActionRead,memy.ActionForget)
 _,e=f.engine.Sweep(t.Context(),f.actor,f.scope,"assist",q);if !errors.Is(e,memy.ErrConflict){t.Fatal(e)}
 q.OperationID="replacement";_,e=f.engine.Sweep(t.Context(),f.actor,f.scope,"assist",q);if !errors.Is(e,memy.ErrRevoked){t.Fatal(e)}
 _,e=f.engine.Fence(t.Context(),f.actor,f.scope,"assist");if !errors.Is(e,memy.ErrRevoked){t.Fatal(e)}
 t.Log("policy bump: original ErrConflict; replacement ErrRevoked; Fence ErrRevoked")
}

}

func TestReviewSweepForeignRevisionKey(t *testing.T){
 f:=newFixture(t,nil); f.config.Retention=reference.Retain[preference]{Version:"retention/v1",ExpiresAt:f.clock.Now().Add(time.Hour)}
 var e error;f.engine,e=memy.New(f.config);if e!=nil{t.Fatal(e)};seedCostCorpus(t,f,1)
 e=f.config.Store.Update(t.Context(),f.scope,func(b memy.Bucket)error{p,e:=b.Scan(memy.ScanOptions{Prefix:"record/",Limit:1,MaxBytes:1<<20});if e!=nil{return e};_,e=b.Put("record/000-foreign/00000000000000000001",0,p.Entries[0].Value.Data);return e});if e!=nil{t.Fatal(e)}
 f.clock.Advance(2*time.Hour)
 for i:=0;i<100;i++{p,e:=f.engine.Sweep(t.Context(),f.actor,f.scope,"assist",memy.SweepRequest{OperationID:"foreign",Limit:128,MaxBytes:1<<20});if e!=nil{t.Fatalf("error %v",e)};if p.Complete{break};if i==99{t.Fatal("not complete")}}
 e=f.config.Store.View(t.Context(),f.scope,func(b memy.Bucket)error{v,e:=b.Get("record/000-foreign/00000000000000000001");if e!=nil{return e};if v.Data==nil{t.Fatal("foreign key erased")};t.Logf("Sweep completed without schema error; foreign canonical row retains %d bytes",len(v.Data));return nil});if e!=nil{t.Fatal(e)}
}
func TestReviewPurgeDoneCorruption(t *testing.T){
 f:=newFixture(t,nil);seedCostCorpus(t,f,1)
 q:=memy.ForgetRequest{OperationID:"review-purge",Selector:memy.Selector{Kind:memy.SelectRecord,ID:"cost-0"},Reason:"review",PolicyVersion:"v1",Limit:1,MaxBytes:1<<20}

 for i:=0;i<3;i++{a,e=f.engine.Forget(t.Context(),f.actor,f.scope,"assist",q);if e!=nil||a.State!=memy.RevocationCommitted{t.Fatal(a,e)}}
 _,e=f.engine.Fence(t.Context(),f.actor,f.scope,"assist");if !errors.Is(e,memy.ErrRevoked){t.Fatal(e)}
