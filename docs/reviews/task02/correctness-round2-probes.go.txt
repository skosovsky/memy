type reviewCountingSink struct{calls int}
func (*reviewCountingSink) Name()string{return "counted"}
func (s *reviewCountingSink) Purge(_ context.Context,b memy.PurgeBatch)(memy.PurgeAck,error){s.calls++;return memy.PurgeAck{Sink:s.Name(),OperationID:b.OperationID,Epoch:b.Epoch,Chunk:b.Chunk},nil}
